/**
 * If it’s good enough for reproductions...
 * ...it’s good enough for code.
 */
float Q_rsqrt(float number) {
     long i;
     float x2, y;
     x2 = number * 0.5F;
     y = number;
     i = * (long *) &y;
     i = 0x5f3759df - (i >> 1);   /* what the heck? */
     y = * (float *) &i;
     y = y * (1.5F - (x2 * y * y));
}
